// 1. 通用大端/小端字节序翻转器 (用于 PPC / MIPS 主机大作的数据结构读取)
// ---------------------------------------------------------------------------
template 
constexpr T EndianSwap(T val) noexcept {
if constexpr (sizeof(T) == 1) {
return val;
} else if constexpr (sizeof(T) == 2) {
uint16_t u = std::bit_cast<uint16_t>(val);
return std::bit_cast(static_cast<uint16_t>((u << 8) | (u >> 8)));
} else if constexpr (sizeof(T) == 4) {
uint32_t u = std::bit_cast<uint32_t>(val);
u = ((u & 0x000000FFu) << 24) | ((u & 0x0000FF00u) << 8) |
((u & 0x00FF0000u) >> 8)  | ((u & 0xFF000000u) >> 24);
return std::bit_cast(u);
} else if constexpr (sizeof(T) == 8) {
uint64_t u = std::bit_cast<uint64_t>(val);
u = ((u & 0x00000000000000FFull) << 56) | ((u & 0x000000000000FF00ull) << 40) |
((u & 0x0000000000FF0000ull) << 24) | ((u & 0x00000000FF000000ull) << 8)  |
((u & 0x000000FF00000000ull) >> 8)  | ((u & 0x0000FF0000000000ull) >> 24) |
((u & 0x00FF000000000000ull) >> 40) | ((u & 0xFF00000000000000ull) >> 56);
return std::bit_cast(u);
}
}

// 自动嗅探归档文件 Magic Number 并分发给对应引擎的解析器
bool UniversalVFSManager::AutoMountArchive(const std::string& mount_point, const std::string& host_archive_path) {
    std::ifstream ifs(host_archive_path, std::ios::binary);
    if (!ifs) {
        std::cout << "[VFS] Ready to mount local archive at: " << host_archive_path << std::endl;
        return false;
    }
    uint32_t magic = 0;
    ifs.read(reinterpret_cast<char*>(&magic), sizeof(magic));
    ifs.close();


    std::unique_ptr<IArchiveParser> parser;
    if (magic == 0x52504637u) {
        parser = std::make_unique<RageRpf7Parser>();
    } else if (magic == 0x58445442u) {
        parser = std::make_unique<CreationBa2Parser>();
    } else if (magic == 0x414B504Bu) {
        parser = std::make_unique<ReEnginePakParser>();
    }


    if (parser && parser->Mount(host_archive_path)) {
        m_mounts[mount_point] = std::move(parser);
        return true;
    }
    return false;
}


bool UniversalVFSManager::LoadAsset(const std::string& virtual_uri, std::vector<uint8_t>& out_data) {
    for (auto& [prefix, parser] : m_mounts) {
        if (virtual_uri.rfind(prefix, 0) == 0) {
            return parser->ReadFile(virtual_uri.substr(prefix.size()), out_data);
        }
    }
    return false;
}

// 净室 YSC 字节码流反汇编器：将字节码还原为结构化指令流，供 REA 与 C++ 转译器消费
std::vector<YscDisasmInstruction> UniversalVMRouter::DisassembleYscCodeblock(const uint8_t* code, size_t size) {
    std::vector<YscDisasmInstruction> out;
    size_t ip = 0;
    while (ip < size) {
        YscDisasmInstruction ins{};
        ins.ip_offset = static_cast<uint32_t>(ip);
        uint8_t op = code[ip++];
        ins.opcode = static_cast<YscOpcode>(op);


        switch (ins.opcode) {
            case YscOpcode::NOP:
                ins.mnemonic = "NOP";
                break;
            case YscOpcode::IADD:
                ins.mnemonic = "IADD";
                break;
            case YscOpcode::PUSH_CONST_U8:
                ins.mnemonic = "PUSH_CONST_U8";
                if (ip < size) ins.operand = code[ip++];
                break;
            case YscOpcode::NATIVE: {
                ins.mnemonic = "CALL_NATIVE";
                if (ip + 2 < size) {
                    uint8_t arg_ret = code[ip++];
                    uint16_t idx = static_cast<uint16_t>((code[ip] << 8) | code[ip + 1]);
                    ip += 2;
                    ins.operand = (static_cast<uint64_t>(arg_ret) << 16) | idx;
                }
                break;
            }
            case YscOpcode::ENTER: {
                ins.mnemonic = "ENTER_FUNC";
                if (ip + 3 < size) {
                    uint8_t param_count = code[ip++];
                    uint16_t frame_vars = static_cast<uint16_t>(code[ip] | (code[ip + 1] << 8));
                    uint8_t name_len = code[ip + 2];
                    ip += 3 + name_len;
                    ins.operand = (static_cast<uint64_t>(param_count) << 16) | frame_vars;
                }
                break;
            }
            case YscOpcode::LEAVE: {
                ins.mnemonic = "LEAVE_FUNC";
                if (ip + 1 < size) {
                    ins.operand = (static_cast<uint64_t>(code[ip]) << 8) | code[ip + 1];
                    ip += 2;
                }
                break;
            }
            default:
                ins.mnemonic = "OP_RAW";
                ins.operand = op;
                break;
        }
        out.push_back(ins);
    }
    return out;
}

def detect_game_engine(game_dir: Path) -> str:
    files_list = [str(p.relative_to(game_dir)) for p in game_dir.rglob("*") if p.is_file()]
    joined = "\n".join(files_list)
    if "GameAssembly.dll" in joined and "global-metadata.dat" in joined:
        return "Unity_IL2CPP"
    if ".utoc" in joined or ("Engine" in joined and ".pak" in joined):
        return "Unreal_Engine_4_5"
    if ".rpf" in joined:
        return "Rockstar_RAGE"
    if ".ba2" in joined or ".esm" in joined:
        return "Bethesda_Creation"
    if "re_chunk_000.pak" in joined:
        return "Capcom_RE_Engine"
    return "Proprietary_Native_Or_Console"

#pragma once
#include "cstdint"
#include "vector"
#include "string"
#include "unordered_map"

namespace uport::gfx {


enum class GuestShaderFormat {
    Unknown,
    DXBC_SM4_SM5,   // PC D3D10 / D3D11 ('DXBC' 0x43425844)
    D3D9_SM3,       // PC / Host D3D9 Vertex/Pixel Shader (0xFFFE0300 / 0xFFFF0300)
    Xenos_Microcode // Xbox 360 Xenos GPU Microcode (0x102A1100)
};


enum class ShaderStage {
    Vertex,
    Pixel,
    Compute,
    Geometry,
    Hull,
    Domain
};


struct TranspiledShaderArtifact {
    uint64_t                shader_hash = 0;
    GuestShaderFormat       source_format = GuestShaderFormat::Unknown;
    ShaderStage             stage = ShaderStage::Vertex;
    std::string             generated_hlsl;
    std::vector< uint32_t > spirv_words;
};


class UniversalShaderRecompiler {
public:
    static UniversalShaderRecompiler& Instance();


    // 计算着色器字节码 64 位 FNV-1a 唯一哈希
    static uint64_t ComputeShaderHash(const void* bytecode, size_t size) noexcept;


    // 嗅探字节码格式并自动转译/命中磁盘与内存 PSO 缓存
    const TranspiledShaderArtifact* GetOrTranspileShader(
        const void* guest_bytecode,
        size_t bytecode_size,
        ShaderStage stage
    );


private:
    GuestShaderFormat DetectFormat(const uint8_t* data, size_t size) const noexcept;
    std::string TranspileDxbcToHlsl(const uint8_t* data, size_t size, ShaderStage stage);
    std::string TranspileXenosToHlsl(const uint8_t* data, size_t size, ShaderStage stage);
    std::vector< uint32_t > CompileHlslToSpirvStub(const std::string& hlsl_src, ShaderStage stage);


    mutable std::mutex m_cache_mutex;
    std::unordered_map< uint64_t, TranspiledShaderArtifact > m_shader_cache;
};


} // namespace uport::gfx
